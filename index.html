De cero absoluto a una aplicación Angular completa

No harás ejemplos sueltos. Construirás una sola app.

Angular Aula será un gestor de estudiantes. Comienza como una carpeta vacía, después se convierte en una interfaz HTML, recibe diseño con CSS y comportamiento con JavaScript, consume una API y finalmente se reconstruye con Angular usando componentes, servicios, formularios, rutas, Signals y HttpClient.

Después agrega los eventos:

app.js · CRUD 
buscar.addEventListener('input', function () {
  const texto = buscar.value.toLowerCase().trim();

  const filtrados = estudiantes.filter(function (estudiante) {
    return estudiante.nombre.toLowerCase().includes(texto) ||
           estudiante.programa.toLowerCase().includes(texto);
  });

  renderizarEstudiantes(filtrados);
});

formulario.addEventListener('submit', function (evento) {
  evento.preventDefault();

  const nuevo = {
    id: Date.now(),
    nombre: formulario.nombre.value.trim(),
    email: formulario.email.value.trim(),
    programa: formulario.programa.value.trim(),
    semestre: Number(formulario.semestre.value)
  };

  estudiantes.push(nuevo);
  formulario.reset();
  renderizarEstudiantes(estudiantes);
});

lista.addEventListener('click', function (evento) {
  if (!evento.target.classList.contains('btn-eliminar')) return;

  const id = Number(evento.target.dataset.id);
  estudiantes = estudiantes.filter(function (estudiante) {
    return estudiante.id !== id;
  });

  renderizarEstudiantes(estudiantes);
});

FASE 1 · JavaScriptFunción: API

Y añade la importación:

app.js · fetch 
async function importarUsuarios() {
  estadoApi.textContent = 'Cargando usuarios...';

  try {
    const respuesta = await fetch('https://jsonplaceholder.typicode.com/users');

    if (!respuesta.ok) {
      throw new Error('HTTP ' + respuesta.status);
    }

    const usuarios = await respuesta.json();

    const importados = usuarios.slice(0, 5).map(function (usuario) {
      return {
        id: Date.now() + usuario.id,
        nombre: usuario.name,
        email: usuario.email,
        programa: 'Importado desde API',
        semestre: 1
      };
    });

    estudiantes = estudiantes.concat(importados);
    renderizarEstudiantes(estudiantes);
    estadoApi.textContent = 'Usuarios importados correctamente.';
  } catch (error) {
    console.error(error);
    estadoApi.textContent = 'No fue posible consultar la API.';
  }
}

btnImportar.addEventListener('click', importarUsuarios);

import { computed, Injectable, signal } from '@angular/core';
import { Estudiante } from '../models/estudiante';

@Injectable({ providedIn: 'root' })
export class Estudiantes {
  private readonly clave = 'angular-aula-estudiantes';
  private readonly _estudiantes = signal<Estudiante[]>(this.leer());

  readonly estudiantes = this._estudiantes.asReadonly();
  readonly total = computed(() => this._estudiantes().length);

  agregar(datos: Omit<Estudiante, 'id'>): void {
    const nuevo: Estudiante = { id: Date.now(), ...datos };
    this._estudiantes.update(actuales => [...actuales, nuevo]);
    this.guardar();
  }

  eliminar(id: number): void {
    this._estudiantes.update(actuales =>
      actuales.filter(estudiante => estudiante.id !== id)
    );
    this.guardar();
  }

  agregarVarios(nuevos: Estudiante[]): void {
    this._estudiantes.update(actuales => [...actuales, ...nuevos]);
    this.guardar();
  }

  private guardar(): void {
    localStorage.setItem(this.clave, JSON.stringify(this._estudiantes()));
  }

  private leer(): Estudiante[] {
    const guardados = localStorage.getItem(this.clave);

    if (guardados) {
      return JSON.parse(guardados);
    }

    return [
      { id: 1, nombre: 'Ana Torres', email: 'ana@correo.com', programa: 'Ingeniería de Sistemas', semestre: 4 },
      { id: 2, nombre: 'Carlos Ruiz', email: 'carlos@correo.com', programa: 'Diseño', semestre: 2 }
    ];
  }
}

  private servicio = inject(EstudiantesService);

  texto = signal('');
  estudiantes = this.servicio.estudiantes;

  filtrados = computed(() => {
    const filtro = this.texto().toLowerCase().trim();

    return this.estudiantes().filter(estudiante =>
      estudiante.nombre.toLowerCase().includes(filtro) ||
      estudiante.programa.toLowerCase().includes(filtro)
    );
  });

  eliminar(id: number): void {
    this.servicio.eliminar(id);
  }
}

import { HttpClient } from '@angular/common/http';
import { inject } from '@angular/core';
import { tap } from 'rxjs';

interface UsuarioApi {
  id: number;
  name: string;
  email: string;
}

// Dentro de la clase:
private http = inject(HttpClient);

importarDesdeApi() {
  return this.http
    .get<UsuarioApi[]>('https://jsonplaceholder.typicode.com/users')
    .pipe(
      tap(usuarios => {
        const nuevos: Estudiante[] = usuarios.slice(0, 5).map(usuario => ({
          id: Date.now() + usuario.id,
          nombre: usuario.name,
          email: usuario.email,
          programa: 'Importado desde API',
          semestre: 1,
          externo: true
        }));

        this.agregarVarios(nuevos);
      })
    );
}
